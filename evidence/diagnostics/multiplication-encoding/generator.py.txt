"""Produce an interned JSON candidate with kernel-checked encoding/readback.

No instruction data is transcribed. The retained projection is the only input.
The generated definitions are candidates; their readback and encoding proofs
are checked by Lean. This prototype writes only under work/.
"""
from __future__ import annotations
import json
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
projection = ROOT / 'outputs/pasta-assembly-verification-bridge/generated/mul-check.json'
target = ROOT / 'work/MulJsonDag.lean'
value = json.loads(projection.read_text(encoding='utf-8'))

def compact(x):
    return json.dumps(x, ensure_ascii=True, separators=(',', ':'))

def literal(x):
    return json.dumps(x, ensure_ascii=True)

nodes = []
interned = {}
keys = {}

def intern(x, fuel):
    k = (compact(x), fuel)
    if k in interned:
        return interned[k]
    children = []
    if isinstance(x, dict):
        for key, child in x.items():
            keys.setdefault(key, len(keys))
            children.append(intern(child, fuel-1))
    elif isinstance(x, list):
        children = [intern(child, fuel-1) for child in x]
    number = len(nodes)
    interned[k] = number
    nodes.append((x, fuel, children))
    return number

root = intern(value, 64)
lines = ['import MulRetained', 'import Init.Data.String.Lemmas', '',
         'namespace PastaBridge.MulJsonDag', '',
         'set_option maxRecDepth 100000', 'set_option maxHeartbeats 0',
         'set_option pp.maxSteps 1000', 'set_option pp.deepTerms false', '',
         'theorem encoded_text_ext {x y : Except String String}',
         '    (h : x.map String.toList = y.map String.toList) : x = y := by',
         '  cases x <;> cases y <;> simp_all [Except.map, String.toList_inj]', '',
         'def charsCheck (x : Except String String) (text : String) : Bool :=',
         '  match x with',
         '  | .error _ => false',
         '  | .ok actual => actual.toList == text.toList', '',
         'theorem encoding_of_chars_check {x : Except String String} {text : String}',
         '    (h : charsCheck x text = true) : x = .ok text := by',
         '  cases x with',
         '  | error reason => simp [charsCheck] at h',
         '  | ok actual =>',
         '    apply congrArg Except.ok',
         '    apply String.toList_injective',
         '    exact eq_of_beq h', '',
         'theorem array_encoding (fuel : Nat) (values : List KernelJson.Value) :',
         '    JsonEncoding.valueText (fuel + 1) (.array values) =',
         '      (values.mapM (JsonEncoding.valueText fuel)).map',
         '        (fun pieces => "[" ++ String.intercalate "," pieces ++ "]") := by rfl', '',
         'theorem object_encoding (fuel : Nat) (fields : List (String × KernelJson.Value)) :',
         '    JsonEncoding.valueText (fuel + 1) (.object fields) =',
         '      (fields.mapM (fun (key, value) => do',
         '        let key ← JsonEncoding.stringText key',
         '        let value ← JsonEncoding.valueText fuel value',
         '        pure (key ++ ":" ++ value))).map',
         '        (fun pieces => "{" ++ String.intercalate "," pieces ++ "}") := by rfl', '']

for key, index in keys.items():
    lines += [f'theorem key_{index} : JsonEncoding.stringText {literal(key)} =',
              f'    .ok {literal(compact(key))} := by decide +kernel', '']

for index, (x, fuel, children) in enumerate(nodes):
    chunks = []
    if isinstance(x, dict):
        fields = ', '.join(f'({literal(key)}, value_{child})'
                           for key, child in zip(x, children))
        expr = f'.object [{fields}]'
    elif isinstance(x, list):
        if len(children) > 64:
            for start in range(0, len(children), 32):
                chunk = children[start:start+32]
                name = f'chunk_{index}_{len(chunks)}'
                chunks.append(name)
                texts = [compact(nodes[child][0]) for child in chunk]
                lines += [f'def {name} : List KernelJson.Value := [' + ', '.join(f'value_{child}' for child in chunk) + ']',
                          f'attribute [local irreducible] {name}',
                          f'theorem mapped_{name} : {name}.mapM (JsonEncoding.valueText {fuel-1}) =',
                          '    .ok [' + ', '.join(literal(text) for text in texts) + '] := by',
                          f'  unfold {name}',
                          '  simp only [' + ', '.join(['List.mapM_cons', 'List.mapM_nil', 'Except.bind',
                              'Except.pure', 'Pure.pure', 'Bind.bind'] +
                              [f'encoded_{child}' for child in dict.fromkeys(chunk)]) + ']', '',
                          f'run_cmd IO.FS.writeFile "mul-json-dag-progress.txt" "checked {name}"', '']
            expr = '.array (' + ' ++ '.join(chunks) + ')'
        else:
            expr = '.array [' + ', '.join(f'value_{child}' for child in children) + ']'
    elif x is None:
        expr = '.null'
    elif isinstance(x, bool):
        expr = '.bool ' + str(x).lower()
    elif isinstance(x, int):
        expr = f'.number ({x})'
    elif isinstance(x, str):
        expr = '.string ' + literal(x)
    else:
        raise ValueError(type(x))
    lines += [f'def value_{index} : KernelJson.Value := {expr}',
              f'attribute [local irreducible] value_{index}', '',
              f'theorem encoded_{index} : JsonEncoding.valueText {fuel} value_{index} =',
              f'    .ok {literal(compact(x))} := by']
    if isinstance(x, (dict, list)):
        lines += ['  apply encoding_of_chars_check']
    lines += [
              f'  unfold value_{index}']
    if isinstance(x, (dict, list)):
        rules = ['object_encoding' if isinstance(x, dict) else 'array_encoding',
                 'List.mapM_cons', 'List.mapM_nil', 'Except.bind', 'Except.pure',
                 'Except.map', 'Bind.bind', 'Pure.pure',
                 'charsCheck', 'String.toList_append', 'String.toList_intercalate']
        if isinstance(x, dict):
            rules += [f'key_{keys[key]}' for key in x]
        if chunks:
            rules += ['List.mapM_append'] + [f'mapped_{name}' for name in chunks]
        else:
            rules += [f'encoded_{child}' for child in dict.fromkeys(children)]
        lines += ['  simp only [' + ', '.join(rules) + ']']
    lines += ['  try rfl' if isinstance(x, (dict, list)) else '  try decide +kernel', '']
    if index % 16 == 0:
        lines += [f'run_cmd IO.FS.writeFile "mul-json-dag-progress.txt" "checked node {index}"', '']

lines += [f'theorem retained_value_readback : retainedMultiplicationArtifactValue = value_{root} := by rfl', '',
          'theorem retained_encoding :',
          '    JsonEncoding.documentText 64 retainedMultiplicationArtifactValue =',
          '      .ok retainedMultiplicationCharonArtifact := by',
          f'  rw [JsonEncoding.documentText, retained_value_readback, encoded_{root}]',
          '  rfl', '',
          '#print axioms retained_value_readback', '#print axioms retained_encoding',
          '', 'end PastaBridge.MulJsonDag', '']
target.write_text('\n'.join(lines), encoding='utf-8')
print(f'{len(nodes)} interned nodes; wrote {target.stat().st_size} bytes')
