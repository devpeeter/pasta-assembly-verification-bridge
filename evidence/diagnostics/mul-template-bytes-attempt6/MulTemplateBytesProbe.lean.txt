import MulMetadata
import StringBytes
import Lean

set_option maxRecDepth 100000
set_option maxHeartbeats 0

namespace PastaBridge

attribute [local irreducible] retainedMultiplicationArtifactValue retainedMultiplicationPieces
  retainedMultiplicationAssembly MultiplicationArtifact.pieces MultiplicationArtifact.renderValue
  MultiplicationArtifact.renderPieces

run_cmd do IO.FS.writeFile "work/mul-template-phases/bytes-start" "before declaration"

theorem template_bytes_probe :
    MultiplicationArtifact.renderValue MultiplicationArtifact.canonicalBinding
      retainedMultiplicationArtifactValue = .ok retainedMultiplicationAssembly := by
  rw [MultiplicationArtifact.renderValue, multiplication_pieces_match_retained_value]
  change Except.ok (MultiplicationArtifact.renderPieces
    MultiplicationArtifact.canonicalBinding retainedMultiplicationPieces) = _
  apply congrArg Except.ok
  apply StringBytes.injective
  rw [MultiplicationArtifact.renderPieces, StringBytes.join]
  run_tac do IO.FS.writeFile "work/mul-template-phases/bytes-list" "after byte observation"
  apply eq_of_beq
  with_unfolding_all rfl
  run_tac do IO.FS.writeFile "work/mul-template-phases/bytes-goals" s!"goals={(← Lean.Elab.Tactic.getGoals).length}"

run_cmd do IO.FS.writeFile "work/mul-template-phases/bytes-done" "after declaration"
#print axioms template_bytes_probe

end PastaBridge
