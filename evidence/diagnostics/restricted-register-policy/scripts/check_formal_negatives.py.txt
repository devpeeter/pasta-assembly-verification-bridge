"""Check kernel-certified addition mutants and reject their false equalities.

Run after building the positive addition correspondence modules. The malformed
proof attempts live in work/, outside the delivered checked proof roots.
"""
from __future__ import annotations

import argparse
import hashlib
import json
import os
from pathlib import Path
import re
import subprocess
import sys

from measure import run

ROOT = Path(__file__).resolve().parents[1]
WORK = ROOT.parent.parent / "work"
NAMESPACE = "PastaBridge.FormalAdditionNegatives."
EXPECTED_DECLARATIONS = {NAMESPACE + name for name in (
    "unmodified_model", "witness_words_bounded", "witness_field_inputs", "expected_value",
    "broken_carry_value", "broken_binding_value", "broken_carry_is_semantically_wrong",
    "broken_binding_is_semantically_wrong")}
ALLOWED_AXIOMS = {"propext", "Classical.choice", "Quot.sound"}
RESULTS = {"broken-carry": "carryResult", "wrong-binding": "bindingResult"}


def check_census(output: str) -> tuple[bool, dict[str, list[str]]]:
    found: dict[str, set[str]] = {}
    duplicate = False
    for name, axioms in re.findall(r"'([^']+)' depends on axioms:\s*\[([^\]]*)\]", output):
        duplicate |= name in found
        found[name] = {value.strip() for value in axioms.split(",") if value.strip()}
    for name in re.findall(r"'([^']+)' does not depend on any axioms", output):
        duplicate |= name in found
        found[name] = set()
    valid = (not duplicate and set(found) == EXPECTED_DECLARATIONS and
             all(values <= ALLOWED_AXIOMS for values in found.values()) and
             not any(marker in output for marker in ("sorryAx", "warning:", "error:")))
    return valid, {name: sorted(values) for name, values in found.items()}


def is_semantic_rejection(result: str, exit_code: int, output: str) -> bool:
    """Accept precisely the pinned Lean diagnostic for the intended equality."""
    if result not in RESULTS.values() or exit_code != 1:
        return False
    diagnostic = output.replace("\r\n", "\n").strip()
    pattern = (r"[^\n]+:\d+:\d+: error: Tactic `decide` proved that the proposition\n  " +
               re.escape(result + " = expected") + r"\nis false")
    return re.fullmatch(pattern, diagnostic) is not None


def lean_environment(lean_bin: Path) -> dict[str, str]:
    env = os.environ.copy()
    env["PATH"] = str(lean_bin) + os.pathsep + env.get("PATH", "")
    suffix = ".exe" if os.name == "nt" else ""
    query = ("import os,json;print(json.dumps({k:os.environ.get(k,'') "
             "for k in ['LEAN_PATH','LEAN_SRC_PATH','PATH']}))")
    env.update(json.loads(subprocess.check_output(
        [str(lean_bin / ("lake" + suffix)), "env", sys.executable, "-c", query],
        cwd=ROOT / "proofs", env=env, text=True)))
    return env


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--lean-bin", type=Path, required=True,
                        help="directory containing the pinned Lean and Lake executables")
    parser.add_argument("--module-dir", type=Path, action="append", default=[],
                        help="additional directory of freshly checked Lean modules")
    args = parser.parse_args()
    lean_bin = args.lean_bin.resolve()
    suffix = ".exe" if os.name == "nt" else ""
    lean = lean_bin / ("lean" + suffix)
    evidence = ROOT / "evidence/formal-addition-negatives"
    evidence.mkdir(parents=True, exist_ok=True)
    build = WORK / "formal-addition-negative-build"
    sources = WORK / "formal-addition-negative-sources"
    build.mkdir(parents=True, exist_ok=True)
    sources.mkdir(parents=True, exist_ok=True)
    helper = ROOT / "proofs/FormalAdditionNegatives.lean"
    report: dict = {"status": "RUNNING", "stages": [],
        "scope": "Closed kernel semantic counterexamples and exact rejection of their false equalities; instruction lists are traversed algorithmically."}

    def save() -> None:
        (evidence / "report.json").write_text(json.dumps(report, indent=2) + "\n", encoding="utf-8")

    save()
    try:
        env = lean_environment(lean_bin)
        env["LEAN_PATH"] = os.pathsep.join([str(build),
            *(str(path.resolve()) for path in args.module_dir), str(WORK / "bridge-build"),
            str(WORK / "kraken-compat/.lake/build/lib/lean"), env.get("LEAN_PATH", "")])
        positive = run([str(lean), "-DautoImplicit=false", "-o", str(build / "FormalAdditionNegatives.olean"), str(helper)],
                       cwd=ROOT / "proofs", env=env, log=evidence / "counterexamples.log")
        output = Path(positive["log"]).read_text(encoding="utf-8", errors="replace")
        census_ok, actual_census = check_census(output)
        positive.update({"status": "PASS" if positive["exit_code"] == 0 and census_ok else "FAIL",
                         "axiom_census": actual_census, "strict_axiom_census_passed": census_ok,
                         "source_sha256": hashlib.sha256(helper.read_bytes()).hexdigest()})
        report["stages"].append({"name": "kernel-counterexample-certificates", **positive})
        save()
        if positive["status"] == "PASS":
            for name, result in RESULTS.items():
                source = sources / (name + ".lean")
                text = ("import FormalAdditionNegatives\n"
                        "open PastaBridge.FormalAdditionNegatives\n"
                        f"example : {result} = expected := by decide +kernel\n")
                source.write_text(text, encoding="utf-8")
                (evidence / (name + ".lean.txt")).write_text(text, encoding="utf-8")
                item = run([str(lean), "-DautoImplicit=false", str(source)],
                           cwd=ROOT / "proofs", env=env, log=evidence / (name + ".log"))
                diagnostic = Path(item["log"]).read_text(encoding="utf-8", errors="replace")
                rejected = is_semantic_rejection(result, item["exit_code"], diagnostic)
                item.update({"status": "PASS" if rejected else "FAIL", "expected_failure": True,
                             "intended_false_equality": result + " = expected",
                             "semantic_rejection_confirmed": rejected,
                             "source_sha256": hashlib.sha256(text.encode()).hexdigest()})
                report["stages"].append({"name": name, **item})
                save()
        report["status"] = "PASS" if len(report["stages"]) == 3 and all(
            stage["status"] == "PASS" for stage in report["stages"]) else "FAIL"
    except (OSError, subprocess.SubprocessError, ValueError) as error:
        report.update({"status": "BLOCKED", "reason": str(error)})
    save()
    print(json.dumps(report, indent=2))
    return 0 if report["status"] == "PASS" else 2 if report["status"] == "BLOCKED" else 1


if __name__ == "__main__":
    raise SystemExit(main())
