"""Adapter policy tests.

The three-operand snippets below are synthetic unit inputs, not transcriptions of
Pasta routines. Integration checks consume only the retained Charon artifact.
"""
import copy
import json
from pathlib import Path
import sys
import tempfile
import unittest

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / "bridge"))
sys.path.insert(0, str(ROOT / "scripts"))
import kraken_adapter as adapter
import extract


def synthetic_block():
    def placeholder(index):
        return {"Placeholder": {"operand_id": index, "modifier": None, "span": None}}
    return {
        "kind": "Asm",
        "options": {name: name in {"pure", "nomem", "nostack"}
                    for name in adapter.OPTION_NAMES},
        "template": [{"Text": "adds "}, placeholder(2), {"Text": ", "},
                     placeholder(0), {"Text": ", "}, placeholder(1)],
        "operands": [
            {"In": {"reg": {"Class": "reg"}, "value": {"synthetic_input": 0}}},
            {"In": {"reg": {"Class": "reg"}, "value": {"synthetic_input": 1}}},
            {"Out": {"reg": {"Class": "reg"}, "late": False, "place": None}}],
    }


class AdapterPolicyTests(unittest.TestCase):
    def test_supported_subset(self):
        block = synthetic_block()
        adapter.validate(block)
        self.assertEqual(adapter.render(block, adapter.canonical_binding(block)),
                         "adds x2, x0, x1")

    def test_equal_read_only_sharing_accepted_unequal_rejected(self):
        block = synthetic_block()
        binding = {0: "x0", 1: "x0", 2: "x2"}
        adapter.validate_binding(block, binding, {0: 7, 1: 7})
        with self.assertRaisesRegex(adapter.BridgeError, "equal input"):
            adapter.validate_binding(block, binding, {0: 7, 1: 8})
        with self.assertRaisesRegex(adapter.BridgeError, "equal input"):
            adapter.validate_binding(block, binding)

    def test_early_output_overlap_rejected_even_when_values_equal(self):
        with self.assertRaisesRegex(adapter.BridgeError, "early output"):
            adapter.validate_binding(synthetic_block(), {0: "x0", 1: "x1", 2: "x0"},
                                     {0: 7, 1: 7})

    def test_contradictory_values_for_same_expression_rejected(self):
        block = synthetic_block()
        block["operands"][1]["In"]["value"] = copy.deepcopy(block["operands"][0]["In"]["value"])
        with self.assertRaises(adapter.BridgeError):
            adapter.validate_binding(block, {0: "x0", 1: "x0", 2: "x2"},
                                     {0: 7, 1: 8})

    def test_reserved_and_incomplete_bindings_rejected(self):
        for register in ("x19", "x29", "sp", "xzr", "w0", "x31"):
            with self.subTest(register=register):
                with self.assertRaisesRegex(adapter.BridgeError, "inadmissible"):
                    adapter.validate_binding(synthetic_block(),
                                             {0: register, 1: "x1", 2: "x2"})
        with self.assertRaisesRegex(adapter.BridgeError, "exactly"):
            adapter.validate_binding(synthetic_block(), {0: "x0"})

    def test_naked_assembly_options_and_directives_rejected(self):
        block = synthetic_block()
        block["kind"] = "NakedAsm"
        with self.assertRaisesRegex(adapter.BridgeError, "only asm"):
            adapter.validate(block)
        for option in adapter.OPTION_NAMES:
            block = synthetic_block()
            block["options"][option] = not block["options"][option]
            with self.subTest(option=option):
                with self.assertRaises(adapter.BridgeError):
                    adapter.validate(block)
        block = synthetic_block()
        block["options"]["unmodeled_option"] = True
        with self.assertRaisesRegex(adapter.BridgeError, "option"):
            adapter.validate(block)
        for text in ("ldr ", ".text\nadds ", "b ", "adds x0; ", "adds [sp], "):
            block = synthetic_block()
            block["template"][0]["Text"] = text
            with self.subTest(text=text):
                with self.assertRaises(adapter.BridgeError):
                    adapter.validate(block)

    def test_unsupported_operand_forms_and_modifiers_rejected(self):
        for variant in ("Const", "SymFn", "SymStatic", "Label"):
            block = synthetic_block()
            block["operands"][0] = {variant: None}
            with self.subTest(variant=variant):
                with self.assertRaisesRegex(adapter.BridgeError, "operand form"):
                    adapter.validate(block)
        block = synthetic_block()
        block["operands"][0]["In"]["reg"] = {"Explicit": "x0"}
        with self.assertRaisesRegex(adapter.BridgeError, "register class"):
            adapter.validate(block)
        block = synthetic_block()
        block["operands"][2]["Out"]["late"] = True
        with self.assertRaisesRegex(adapter.BridgeError, "lateout"):
            adapter.validate(block)
        block = synthetic_block()
        block["template"][1]["Placeholder"]["modifier"] = "w"
        with self.assertRaisesRegex(adapter.BridgeError, "modifiers"):
            adapter.validate(block)

    def test_arbitrary_incoming_flags_and_scratch_not_read(self):
        block = synthetic_block()
        block["template"][0]["Text"] = "adcs "
        with self.assertRaisesRegex(adapter.BridgeError, "incoming carry"):
            adapter.validate(block)
        block = synthetic_block()
        block["template"][3]["Placeholder"]["operand_id"] = 2
        with self.assertRaises(adapter.BridgeError):
            adapter.validate(block)

    def test_literal_register_is_rejected(self):
        block = synthetic_block()
        block["template"].append({"Text": "\nadds x2, x0, x1"})
        with self.assertRaisesRegex(adapter.BridgeError, "literal.*register|physical.*register"):
            adapter.validate(block)

    def test_immediate_shift_grammar_and_bounds_match_parser_boundary(self):
        for opcode in ("lsl", "lsr"):
            for amount in (0, 2, 62, 63):
                adapter._instruction(f"{opcode} x0, x1, #{amount}", 1)
            for operand in ("#64", "#-1", "#0x2", "#0b10", "#label", "x2", "#184467440737095516160000"):
                with self.subTest(opcode=opcode, operand=operand):
                    with self.assertRaises(adapter.BridgeError):
                        adapter._instruction(f"{opcode} x0, x1, {operand}", 1)


class RetainedArtifactTests(unittest.TestCase):
    def setUp(self):
        self.artifact = ROOT / "artifacts/charon-asm.json"
        if not self.artifact.is_file():
            self.skipTest("genuine Charon extraction has not been produced")

    def test_complete_subtrees_match_raw_artifact(self):
        artifact = adapter.load_artifact(self.artifact)
        extract.verify_retained(artifact, self.artifact.parent)
        counts = {name: len([line for line in adapter.render(block,
                    adapter.canonical_binding(block)).splitlines() if line.strip()])
                  for name, block in artifact["routines"].items()}
        self.assertEqual(counts, {"add": 12, "mul": 135})

    def test_raw_or_operand_tampering_rejected(self):
        artifact = adapter.load_artifact(self.artifact)
        with tempfile.TemporaryDirectory() as temp:
            root = Path(temp)
            raw_name = artifact["extraction"]["raw_artifact"]
            raw = (self.artifact.parent / raw_name).read_bytes()
            (root / raw_name).write_bytes(raw + b" ")
            with self.assertRaisesRegex(ValueError, "SHA-256"):
                extract.verify_retained(artifact, root)
            (root / raw_name).write_bytes(raw)
            for name in ("add", "mul"):
                changed = copy.deepcopy(artifact)
                ops = changed["routines"][name]["operands"]
                if isinstance(ops, dict):
                    ops = ops["vector"]
                ops[0], ops[1] = ops[1], ops[0]
                with self.subTest(routine=name), self.assertRaisesRegex(ValueError, "envelope differs"):
                    extract.verify_retained(changed, root)

    def test_real_generated_file_staleness_rejected(self):
        from integrity import compare_generated, IntegrityError
        expected = adapter.generate(self.artifact)
        with tempfile.TemporaryDirectory() as temp:
            root = Path(temp)
            for name, content in expected.items():
                (root / name).write_bytes(content)
            compare_generated(expected, root)
            for name in ("add.S", "mul.S", "mul-check.json"):
                path = root / name
                if name.endswith(".S"):
                    path.write_bytes(expected[name].replace(b"adcs", b"adds", 1))
                else:
                    path.write_bytes(expected[name].replace(b'"late":false', b'"late":true', 1))
                with self.subTest(artifact=name), self.assertRaisesRegex(IntegrityError, "stale generated"):
                    compare_generated(expected, root)
                path.write_bytes(expected[name])


if __name__ == "__main__":
    unittest.main()
