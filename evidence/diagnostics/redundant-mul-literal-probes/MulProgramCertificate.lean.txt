import MulMetadata
import Init.Data.String.Lemmas.Intercalate

open Lean Elab Term

set_option maxRecDepth 100000
set_option maxHeartbeats 0

namespace PastaBridge

theorem multiplication_template_matches_retained_assembly :
    MultiplicationArtifact.renderValue MultiplicationArtifact.canonicalBinding
      retainedMultiplicationArtifactValue = .ok retainedMultiplicationAssembly := by
  rw [MultiplicationArtifact.renderValue, multiplication_pieces_match_retained_value]
  change Except.ok (MultiplicationArtifact.renderPieces
    MultiplicationArtifact.canonicalBinding retainedMultiplicationPieces) = _
  apply congrArg Except.ok
  apply String.toList_injective
  rw [MultiplicationArtifact.renderPieces, String.toList_join]
  rfl

def multiplicationSerializedLines (program : Program) : Except String (List String) :=
  program.mapM fun directive => do
    match directive with
    | .instr ⟨.W64, operation⟩ => MultiplicationArtifact.serializeOperation operation
    | _ => throw "unsupported directive or instruction width"

elab "quotedMultiplicationSerializedLines" : term => do
  match multiplicationSerializedLines retainedMultiplicationProgram with
  | .ok lines => return Lean.toExpr lines
  | .error reason => throwError reason

/-- A candidate serialization of the actual native instructions, checked below. -/
def retainedMultiplicationSerializedLines : List String := quotedMultiplicationSerializedLines

theorem multiplication_serialized_lines_readback :
    multiplicationSerializedLines retainedMultiplicationProgram =
      .ok retainedMultiplicationSerializedLines := by rfl

theorem multiplication_serialization_is_lines (program : Program) :
    MultiplicationArtifact.serializeProgram program =
      (multiplicationSerializedLines program).map (String.intercalate "\n") := rfl

theorem multiplication_program_serializes_to_retained_assembly :
    MultiplicationArtifact.serializeProgram retainedMultiplicationProgram =
      .ok retainedMultiplicationAssembly := by
  rw [multiplication_serialization_is_lines, multiplication_serialized_lines_readback]
  change Except.ok (String.intercalate "\n" retainedMultiplicationSerializedLines) = _
  apply congrArg Except.ok
  apply String.toList_injective
  rw [String.toList_intercalate]
  rfl

theorem multiplication_artifact_matches_retained_program :
    MultiplicationArtifact.renderValue MultiplicationArtifact.canonicalBinding
      retainedMultiplicationArtifactValue =
      MultiplicationArtifact.serializeProgram retainedMultiplicationProgram := by
  rw [multiplication_template_matches_retained_assembly,
    multiplication_program_serializes_to_retained_assembly]

#print axioms multiplication_template_matches_retained_assembly
#print axioms multiplication_program_serializes_to_retained_assembly
#print axioms multiplication_artifact_matches_retained_program

end PastaBridge
