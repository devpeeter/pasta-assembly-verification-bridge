import AdditionNatObservation
import PastaCurves.AArch64.Transcription

namespace PastaBridge.AdditionNatData

set_option maxRecDepth 100000
set_option maxHeartbeats 0
set_option pp.maxSteps 1000
set_option pp.deepTerms false
set_option pp.deepTerms.threshold 30

def output (state : AdditionNat.State) : PastaCurves.Limbs :=
  ⟨state.words 0, state.words 1, state.words 2, state.words 3⟩
def rhs (state : AdditionNat.State) : PastaCurves.Limbs :=
  ⟨state.words 4, state.words 5, state.words 6, state.words 7⟩
def modulus (state : AdditionNat.State) : PastaCurves.Limbs :=
  ⟨state.words 8, state.words 9, 0, state.words 10⟩

attribute [local irreducible] sequence AdditionNat.transition AdditionNat.update AdditionNat.read
attribute [local irreducible] PastaCurves.addc PastaCurves.AArch64.subc PastaCurves.AArch64.cselCs

theorem retained_data_model (state : AdditionNat.State) :
    output (sequence retainedData state) =
      PastaCurves.AArch64.addMod (output state) (rhs state) (modulus state) := by
  run_tac do
    IO.FS.writeFile "addition-model-before-simp.txt" "entered simp"
    Lean.Elab.Tactic.evalTactic (← `(tactic|
      simp (ground := true) only [↓output, ↓rhs, ↓modulus, retainedData,
    ↓words_sequence_cons, ↓words_sequence_nil, ↓carry_sequence_cons, ↓carry_sequence_nil,
    ↓AdditionNat.words_transition, ↓AdditionNat.carry_transition,
    ↓AdditionNat.read_operand, ↓AdditionNat.read_zero, ↓reduceIte,
    PastaCurves.AArch64.addMod, Fin.mk.injEq, if_true, if_false]))
    IO.FS.writeFile "addition-model-after-simp.txt" "simp returned"

run_cmd do
  IO.FS.writeFile "addition-model-after-declaration.txt" "declaration checked"

#print axioms retained_data_model

end PastaBridge.AdditionNatData
