theorem multiplication_template_matches_retained_assembly :
    MultiplicationArtifact.renderValue MultiplicationArtifact.canonicalBinding
      retainedMultiplicationArtifactValue = .ok retainedMultiplicationAssembly := by
  rw [MultiplicationArtifact.renderValue, multiplication_pieces_match_retained_value]
  change Except.ok (MultiplicationArtifact.renderPieces
    MultiplicationArtifact.canonicalBinding retainedMultiplicationPieces) = _
  apply congrArg Except.ok
  apply String.toList_injective
  rw [MultiplicationArtifact.renderPieces, String.toList_join]
  rfl

def multiplicationSerializedLines (program : Program) : Except String (List String) :=
  program.mapM fun directive => do
    match directive with
    | .instr ⟨.W64, operation⟩ => MultiplicationArtifact.serializeOperation operation
    | _ => throw "unsupported directive or instruction width"

elab "quotedMultiplicationSerializedLines" : term => do
  match multiplicationSerializedLines retainedMultiplicationProgram with
  | .ok lines => return Lean.toExpr lines
  | .error reason => throwError reason

/-- A candidate serialization of the actual native instructions, checked below. -/
def retainedMultiplicationSerializedLines : List String := quotedMultiplicationSerializedLines

    MultiplicationArtifact.serializeProgram retainedMultiplicationProgram =
      .ok retainedMultiplicationAssembly := by
