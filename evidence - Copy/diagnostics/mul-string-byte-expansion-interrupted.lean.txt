import MulRetained
import StringBytes
import Lean

open Lean Elab Term

namespace PastaBridge.MulRetainedBytes

set_option maxRecDepth 1000000
set_option maxHeartbeats 0
set_option pp.maxSteps 1000
set_option pp.deepTerms false

/-- A candidate byte list quoted from the already retained file literal. -/
elab "quotedRetainedMultiplicationBytes" : term =>
  return Lean.toExpr (StringBytes.bytes retainedMultiplicationCharonArtifact)

def retainedBytes : List UInt8 := quotedRetainedMultiplicationBytes

/-- This is a kernel readback goal, never justified by the elaborator evaluation. -/
theorem retained_bytes_readback :
    StringBytes.bytes retainedMultiplicationCharonArtifact = retainedBytes := by
  run_tac
    IO.FS.writeFile "mul-retained-bytes-before.txt"
      ("goals=" ++ toString (← Lean.Elab.Tactic.getGoals).length)
  unfold retainedBytes
  rfl
  run_tac
    IO.FS.writeFile "mul-retained-bytes-after.txt"
      ("goals=" ++ toString (← Lean.Elab.Tactic.getGoals).length)

run_cmd IO.FS.writeFile "mul-retained-bytes-after-declaration.txt" "accepted"

#print axioms retained_bytes_readback

end PastaBridge.MulRetainedBytes
