import KrakenWord
import Mathlib.Tactic.SimpRw

namespace PastaBridge.NormalizationProbe

example (s : Reg64s) :
    s.X4.toBitVec <<< (Int64.toInt (0 : Int64)).toNat = s.X4.toBitVec := by
  have hz : (Int64.toInt (0 : Int64)).toNat = 0 := by decide +kernel
  rw [hz, BitVec.shiftLeft_zero]

example (flags : StatusFlags) (x y : BitVec 64) :
    (if CondCode.CS.interp flags then x else y).toNat =
      (if flags.c.toNat = 0 then y.toNat else x.toNat) := by
  rw [KrakenWord.csel_cs_value]

example (s : Reg64s) (flags : StatusFlags) (x y : BitVec 64) :
    ((s.setXReg .X0 (if CondCode.CS.interp flags then x else y)).X0).toNat =
      (if flags.c.toNat = 0 then y.toNat else x.toNat) := by
  simp only [Reg64s.setXReg, UInt64.toNat_ofBitVec]
  dsimp (instances := true) only [Width.type, Width.bits]
  simp only [KrakenWord.csel_cs_value]

example (s : Reg64s) :
    ShiftRegExpr.interp (w := .W64)
      { reg := .low (.reg .X4) .W64, shift := .LSL, amount := 0 }
      s ⟨0, 4⟩ = s.X4.toBitVec := by
  have hz : (Int64.toInt (0 : Int64)).toNat = 0 := by decide +kernel
  simp only [ShiftRegExpr.interp, Reg64s.getRegOrZr, Reg64s.getRegOrZr64,
    Reg64s.getXReg, RegOrZr.base, BitVec.take, Width.bits,
    BitVec.extractLsb'_eq_self, hz]
  dsimp (instances := true) only [Width.type, Width.bits]
  simp only [BitVec.shiftLeft_zero]

end PastaBridge.NormalizationProbe
